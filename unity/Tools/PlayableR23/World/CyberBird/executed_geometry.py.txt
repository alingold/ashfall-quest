def _beam(a, b, thickness, material, solid=False):
    axis = b - a
    if axis.length() < 0.001:
        return
    basis = Basis.looking_at(axis.normalized(), Vector3.FORWARD if absf(axis.normalized().y) > 0.99 else Vector3.UP)
    _box((a + b) * 0.5, Vector3(thickness, thickness, axis.length()), material, solid, False, basis)

def _facade(center, span, perpendicular, y, side, rotated, mat, id, level):
    bays = maxi(2, int(gd_div(span, 6)))
    step = gd_div(span, bays)
    for i in gd_iter(bays):
        u = -span * 0.5 + step * (i + 0.5)
        door = i == gd_div(bays, 2) or (level == 0 and i == gd_div(bays, 2) - 1)
        if not door and gd_mod(id + level + i + (1 if rotated else 0), 3) == 0:
            _box(_swap(Vector3(u, y + 2.1, side * perpendicular * 0.5), rotated), _swap(Vector3(step, 4.2, 0.48), rotated), mat)
            continue
        sill = 0.0 if door else 0.92
        pos = Vector3(u, y + sill * 0.5, side * perpendicular * 0.5)
        size = Vector3(step, sill, 0.32)
        if sill > 0:
            _box(_swap(pos, rotated), _swap(size, rotated), mat)
        _box(_swap(Vector3(u, y + 3.85, side * perpendicular * 0.5), rotated), _swap(Vector3(step, 0.7, 0.32), rotated), mat)
        _box(_swap(Vector3(-span * 0.5 + i * step, y + 2.1, side * perpendicular * 0.5), rotated), _swap(Vector3(0.48, 4.2, 0.48), rotated), 'metal')
        local = _swap(Vector3(u, y + 2, side * perpendicular * 0.5), rotated)
        windows.append(Row({'at': global_position + center + local, 'normal': _swap(Vector3(0, 0, side), rotated), 'building': id, 'floor': level, 'door': door, 'width': step - 0.48, 'sill': sill}))
    _box(_swap(Vector3(span * 0.5, y + 2.1, side * perpendicular * 0.5), rotated), _swap(Vector3(0.48, 4.2, 0.48), rotated), 'metal')

def _swap(v, rotated):
    return Vector3(v.z, v.y, v.x) if rotated else v

def _nest(at, radius, accent):
    for i in gd_iter(12):

def _bridge(a, b, accent, index):
    _begin_chunk(gd_mod('SkyBridge_%02d', index))
    direction = (b - a).normalized()
    length = a.distance_to(b)
    basis = Basis.looking_at(direction, Vector3.UP)
    mid = (a + b) * 0.5
    _box(mid - Vector3.UP * 0.15, Vector3(3.6, 0.3, length), 'metal', True, False, basis)
    for side in gd_iter(Array([-1.0, 1.0])):
        _box(mid + basis.x * side * 1.73 + Vector3.UP * 0.55, Vector3(0.16, 1.1, length), 'metal', True, False, basis)
        _box(mid + basis.x * side * 1.75 + Vector3.UP * 1.14, Vector3(0.08, 0.08, length), accent, False, False, basis)

def _loot_offset_r6(b, slot):
    w = b.width
    d = b.depth
    match slot:
        case 0:
            return Vector3(-w * 0.28, 1.45, d * 0.5 + 0.55)
        case 1:
            return Vector3(w * 0.28, 1.45, d * 0.5 + 0.55)
        case 2:
            return Vector3(w * 0.5 + 0.55, 1.45, -d * 0.25)
        case 3:
            return Vector3(-w * 0.24, 0.75, -d * 0.22)
        case 4:
            return Vector3(-w * 0.24, STOREY + 0.75, -d * 0.22)
    return Vector3(-w * 0.24, int(b.floors) * STOREY + 0.75, -d * 0.22)
